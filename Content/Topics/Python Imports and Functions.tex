Python Standard Library:
\begin{enumerate}
	\item math (numpy is better)
	\begin{enumerate}
		\item sin, cos, tan (asin, sinh, ...)
		\item exp, log, erf
		\item ceil, floor
		\item isinf, isnan
		\item degrees, radians
		\item pi, e, inf, nan
	\end{enumerate}
	\item statistics (numpy is better)
	\begin{enumerate}
		\item mean, median, mode
		\item stdev, variance, covariance, quantile
		\item linear\_regression
	\end{enumerate}
	\item random (numpy is also good)
	\begin{enumerate}
		\item seed
		\item randint, randrange
		\item random, uniform
		\item choice, sample
		\item shuffle
	\end{enumerate}
	\item pickle
	\begin{enumerate}
		\item dump, load
	\end{enumerate}
	\item csv (just use numpy)
	\item threading, multiprocessing (parallel processing)
	\item time
	\begin{enumerate}
		\item sleep
		\item time
	\end{enumerate}
	\item os
	\begin{enumerate}
		\item listdir, getcwd, mkdir
	\end{enumerate}
	\item sys
	\begin{enumerate}
		\item argc, argv
	\end{enumerate}
\end{enumerate}

Third-party Libraries:

	\item numpy
	\begin{enumerate}
		\item everything from math, statistics, and random and more (sometimes with different names)
	\end{enumerate}
	\item scipy
	\item scikit-learn
	\item pandas
	\item seaborn
\end{enumerate}
